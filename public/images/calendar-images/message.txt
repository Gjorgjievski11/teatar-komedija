<script>
  document.addEventListener('DOMContentLoaded', function () {
    const monthSelector = new Swiper('.month-selector', {
      slidesPerView: 5,
      centeredSlides: true,
      spaceBetween: 10,
      loop: true,
      speed: 500,
      navigation: {
        nextEl: '.month-selector-button-next',
        prevEl: '.month-selector-button-prev',
      },
      on: {
        init: function () {
          const centerIndex = Math.floor(this.slides.length / 2);
          this.slides[centerIndex].classList.add('text-red-500', 'font-bold', 'scale-110');
        },
        slideChange: function () {
          this.slides.forEach(slide => {
            slide.classList.remove('text-red-500', 'font-bold', 'scale-110');
          });

          const centerIndex = this.activeIndex + 2; // 3rd visible slide (0-indexed + 2)
          const centerSlide = this.slides[centerIndex];
          if (centerSlide) {
            centerSlide.classList.add('text-red-500', 'font-bold', 'scale-110');
          }
        }
      }
    });
  });
</script>
